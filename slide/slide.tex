\section{Introduction}
\subsection{Overview}

\begin{frame}{Introduction - Overview}
    We have implemented a high-performance ray tracing rendering system purely on GPU (DirectX12) from scratch. Unlike traditional approaches that rely on pre-existing frameworks, building the entire pipeline ourselves ensures maximum control and optimization. Additionally, our renderer supports a variety of special effects.
\end{frame}

\subsection{Technics}

\begin{frame}{Introduction - Technics}
    \begin{itemize}
        \item Multiple Formats Support: OBJ, glTF, FBX, Blend
        \item GPU-based Path Tracing: Acceleration Structures, Material Evaluation, Light Sampling, Virtual Textures
        \item Denoising: Intel Open Image Denoise Integration\footnote{https://www.openimagedenoise.org/}
        \item Other Special Effects: Depth of Field, Chromatic Dispersion, Relativity Effect
    \end{itemize}
\end{frame}

\subsection{Relativistic Rendering}
\begin{frame}{Relativistic Rendering - Theoretical Background}
    \textbf{Three Visual Effects:}
    \begin{itemize}
        \item \textbf{Aberration}: Light direction changes, field of view shrinks forward.
        \item \textbf{Doppler Shift}: Color frequency changes (Blue shift/Red shift).
        \item \textbf{Headlight Effect}: Brightness increases forward, dims backward.
    \end{itemize}
    \textbf{Lorentz Transformation:}
    $$ \beta = v/c, \quad \gamma = \frac{1}{\sqrt{1 - \beta^2}} $$
    $$ \Delta t' = \gamma \Delta t, \quad L' = L/\gamma $$
    Significant effects appear when $\beta \to 1$.
\end{frame}

\begin{frame}{Relativistic Rendering - Aberration}
    \textbf{Theory:} $\cos(\theta') = \frac{\cos(\theta) - \beta}{1 - \beta \cos(\theta)}$ (Inverse transform for ray tracing)
    
    \textbf{Implementation:}
    \begin{itemize}
        \item Calculate the angle $\theta'$ between the view direction and velocity.
        \item Apply the inverse aberration formula to find the original angle $\theta$.
        \item Construct the new ray direction using the computed $\theta$.
    \end{itemize}
    
    \textbf{Pseudocode:}
    \begin{itemize}
        \item $v_{hat} \leftarrow velocity / \beta$
        \item $\cos\theta' \leftarrow dot(dir, v_{hat})$
        \item $\cos\theta \leftarrow (\cos\theta' - \beta) / (1 - \beta \cdot \cos\theta')$
        \item $newDir \leftarrow reconstruct(v_{hat}, \cos\theta)$
    \end{itemize}
\end{frame}

\begin{frame}{Relativistic Rendering - Doppler Effect}
    \textbf{Theory:} $D = \gamma(1 + \beta \cos(\theta))$, $f' = D \cdot f$
    
    \textbf{Factor Calculation:}
    \begin{itemize}
        \item Determine the angle between light direction and velocity.
        \item Compute the Doppler factor $D$ using the Lorentz factor $\gamma$.
    \end{itemize}
    
    \textbf{Pseudocode:}
    \begin{itemize}
        \item $lightDir \leftarrow -rayDir$
        \item $\cos\theta \leftarrow dot(lightDir, v_{hat})$
        \item $D \leftarrow \gamma \cdot (1 + \beta \cdot \cos\theta)$
    \end{itemize}
\end{frame}

\begin{frame}{Relativistic Rendering - Color Shift}
    \textbf{Applying Doppler Shift:}
    \begin{itemize}
        \item Shift RGB wavelengths: $\lambda' = \lambda / D$.
        \item Apply intensity falloff for wavelengths moving out of the visible spectrum.
        \item Convert shifted wavelengths back to RGB colors.
    \end{itemize}
    
    \textbf{Pseudocode:}
    \begin{itemize}
        \item $wavelengths \leftarrow \{650, 550, 450\}$ (RGB)
        \item $shiftedWavelengths \leftarrow wavelengths / D$
        \item $falloff \leftarrow calculateVisibleFalloff(shiftedWavelengths)$
        \item $resultColor \leftarrow wavelengthToRGB(shiftedWavelengths) \cdot originalColor \cdot falloff$
    \end{itemize}
\end{frame}

\begin{frame}{Relativistic Rendering - Headlight Effect}
    \textbf{Theory:} $I' = I \times D^n$ (We use $n=2$ for balance).
    
    \textbf{Implementation:}
    \begin{itemize}
        \item Calculate Doppler factor $D$ for the view direction.
        \item Scale the environment light intensity by $D^2$.
        \item Clamp the factor to avoid extreme brightness.
    \end{itemize}
    
    \textbf{Pseudocode:}
    \begin{itemize}
        \item $D \leftarrow RelativisticDopplerFactor(viewDir, velocity)$
        \item $headlightFactor \leftarrow clamp(D^2, 0.1, 10.0)$
        \item $finalRadiance \leftarrow envRadiance \cdot headlightFactor$
    \end{itemize}
\end{frame}

\begin{frame}{Relativistic Rendering - Indirect Lighting}
    \textbf{Challenge:} Indirect light bounces multiple times.
    
    \textbf{Solution (Unified Doppler):} The frequency shift depends on the initial observation direction relative to velocity.
    
    \textbf{Implementation:}
    \begin{itemize}
        \item Store the initial ray direction in the ray payload.
        \item For all subsequent bounces (indirect lighting), use the stored initial direction to calculate the Doppler factor.
        \item This ensures consistent frequency shifting across the entire light path.
    \end{itemize}
\end{frame}
